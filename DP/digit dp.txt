using namespace std;
const char D = 52;
const int MOD = 1e9 + 7;
int dp[D][2][2][36][36][2];
char vis[D][2][2][35][35][2];
char tc = 0;
string a, b;
string normlen(string &s, char sz) { return s = (string(sz - (char)s.size(), '0') + s); }
int add(int a, int b){
    a += b;
    return (a >= MOD ? a - MOD : a);
}
int rec(char i, bool upl, bool downh, char trs, char ns, bool vld){
    auto &ret = dp[i][upl][downh][trs][ns][vld];
    if (i == (int)b.size())
        return ((trs == 17) && (ns == 17) && vld);
    auto &vz = vis[i][upl][downh][trs][ns][vld];
    if (vz == tc)
        return ret;
    vz = tc, ret = 0;
    char low = (upl ? 0 : a[i]);
    char high = (downh ? 9 : b[i]);
    for (char curv, curt, curs, curn, j = low; j <= high; j++)
    {
        curt = (j == 3);
        curs = (j == 6);
        curn = (j == 9);
        if ((curs && (ns == 0 || trs == 0)) || curt && trs > 34 || curn && ns > 34)
            continue;
        ret = add(ret, rec(i + 1, (upl || j > low), (downh || j < high), trs + curt - curs, ns + curn - curs, vld || (max({curt, curs, curn}))));
    }
    return ret;
}
void solve(){
    cin >> a >> b;
    normlen(a, b.size());
    for (auto &nm : a)
        nm -= '0';
    for (auto &nm : b)
        nm -= '0';
    cout << rec(0, 0, 0, 17, 17, 0) << '\n';
}
signed main(){
    fio int t;
    cin >> t;
    for (tc = 1; tc <= t; tc++)
        solve();
    return 0;
}
