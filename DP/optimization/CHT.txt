#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
#define el '\n'
#define fio ios_base::sync_with_stdio(0); cin.tie(0); cout.tie(0);
//#pragma once
// to find the minimum
// add(-k, -m)
// query(x) * -1
// max(mx+b), query(x), intercept is what's added
// add(slope, intercept) (slope*xi) which is queried
struct Line {
    ll k, m;
    mutable ll p;
    bool operator<(const Line& o) const { return k < o.k;}
    bool operator<(const ll&x) const { return p < x;}
};

struct LineContainer : multiset<Line, less<>> {
    const ll inf = numeric_limits<ll>::max();
    ll div(ll a, ll b) { // floored division
        return a / b - ((a ^ b) < 0 && a % b);
    }
    bool isect(iterator x, iterator y) {
        if (y == end()) { x->p = inf; return false; }
        if (x->k == y->k) x->p = x->m > y->m ? inf : -inf;
        else x->p = div(y->m - x->m, x->k - y->k);
        return x->p >= y->p;
    }
    void add(ll k, ll m) {
        auto z = insert({k, m, 0}), y = z++, x = y;
        while (isect(y, z)) z = erase(z);
        if (x != begin() && isect(--x, y)) isect(x, y = erase(y));
        while ((y = x) != begin() && (--x)->p >= y->p)
            isect(x, erase(y));
    }
    ll query(ll x) {
        assert(!empty());
        auto l = *lower_bound(x);
        return l.k * x + l.m;
    }
};

void solve() {
    LineContainer cht;
    ll n, c, lst{};
    cin >> n >> c;
    for (int i = 0; i < n; i++) {
        ll h;
        cin >> h;
        if (i)
            lst = cht.query(h)-h*h-c, lst*=-1;
        cht.add(2*h,-lst-h*h);
    }
    cout << lst << el;
}

signed main() { fio int t = 1; /*cin >> t;*/ while (t--) solve(); return 0; }
