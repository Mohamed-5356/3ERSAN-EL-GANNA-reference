#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
// inf, op<=, best >
#define RECONSTRUCT 1          // 0 = value only (O(n) memory), 1 = also recover splits (O(L*n))

const ll INF = LLONG_MAX / 4;

int n, L;
vector<ll> a, pre;             // 1-indexed input + prefix sums
vector<ll> prv, cur;
#if RECONSTRUCT
vector<vector<int>> opt;       // opt[layer][j] = best split k for dp[layer][j]
#endif

// cost of segment a[k+1..j]; must be O(1)
inline ll C(int k, int j) {
    ll s = pre[j] - pre[k];
    return s * s;              // <-- problem-specific
}

// fill cur[lo..hi]; optimal split for each lies in [optl, optr]
void solve(int lo, int hi, int optl, int optr, int layer) {
    if (lo > hi) return;
    int mid = (lo + hi) >> 1;
    ll best = INF;
    int o = optl;
    for (int k = optl, kend = min(mid - 1, optr); k <= kend; k++) {
        if (prv[k] >= INF) continue;
        ll v = prv[k] + C(k, mid);
        if (v < best) { best = v; o = k; }
    }
    cur[mid] = best;
#if RECONSTRUCT
    opt[layer][mid] = o;
#endif
    solve(lo, mid - 1, optl, o, layer);
    solve(mid + 1, hi, o, optr, layer);
}

int main() {
    ios::sync_with_stdio(false); cin.tie(nullptr);
    cin >> n >> L;
    a.assign(n + 1, 0);
    pre.assign(n + 1, 0);
    for (int i = 1; i <= n; i++) { cin >> a[i]; pre[i] = pre[i - 1] + a[i]; }

    prv.assign(n + 1, INF);
    cur.assign(n + 1, INF);
    prv[0] = 0;
#if RECONSTRUCT
    opt.assign(L + 1, vector<int>(n + 1, 0));
#endif

    for (int layer = 1; layer <= L; layer++) {
        fill(cur.begin(), cur.end(), INF);
        solve(1, n, 0, n - 1, layer);
        prv.swap(cur);
    }

    if (prv[n] >= INF) { cout << "-1\n"; return 0; }   // infeasible: L groups can't cover n
    cout << prv[n] << "\n";

#if RECONSTRUCT
    vector<pair<int,int>> groups;
    int j = n;
    for (int layer = L; layer >= 1; layer--) {
        int k = opt[layer][j];
        groups.push_back({k + 1, j});
        j = k;
    }
    reverse(groups.begin(), groups.end());
    for (auto &[l, r] : groups) cout << l << " " << r << "\n";
#endif
}
