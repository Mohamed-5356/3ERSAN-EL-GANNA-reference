#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
#define el '\n'
#define fio ios_base::sync_with_stdio(0); cin.tie(0); cout.tie(0);
const int N = 200123;
vector<int> g[N];
int sz[N], n;
bool vis[N];
int getsz(int u, int p = -1) {
    if (vis[u]) return 0;
    sz[u] = 1;
    for (int v : g[u])
        if (v != p) sz[u] += getsz(v,u);
    return sz[u];
}
int getc(int u, int p, int n) {
    for (int v : g[u])
        if (v != p && !vis[v] && sz[v] > n/2)
            return getc(v,u, n);
    return u;
}
int id = 0;
//vector<set<pair<int, int>>> ans[N];
//int bid[N*21];
vector<tuple<int, int, int>> dist[N];
priority_queue<pair<int, int>> bpq[N*20];
pair<int, int> bcur[N*20];
set<tuple<int, int, int>> chck[N];
bool vis2[N];
void initc(int u = 0, int p = -1) {
    getsz(u);
    int c = getc(u,-1,sz[u]);
    queue<tuple<int,int, int>> q; q.push({c,-1, 0});
    vector<int> vc;
    vis[c] = 1;
    while (q.size()) {
        auto [u,f, lv] = q.front();
        q.pop();
        for (auto &v : g[u]) {
            if (!vis[v] && !vis2[v]) {
                int nwf =  (f == -1? (++id) : f);
                q.push({v, nwf, lv + 1});
                vis2[v] = 1;
                vc.push_back(v);
                dist[v].push_back({c, nwf, lv + 1});
            }
        }
    }
    for (auto i : vc) vis2[i] = 0;
    vc.clear();
    for (int v : g[c])
        if (!vis[v])
            initc(v,c);
}
int cl[N*20];
pair<int, int> bmax(int bid) {
    while (bpq[bid].size() && cl[bpq[bid].top().second] == 1)
        bpq[bid].pop();
    return bpq[bid].size()? bpq[bid].top() : make_pair(-1, -1);
}
void sync(int c, int bid) {
    auto cur = bcur[bid];
    auto nw = bmax(bid);
    if (nw == cur) return;
    if (cur.first!=-1) chck[c].erase({cur.first, cur.second, bid});
    cur = nw;
    if (cur.first!=-1) chck[c].insert({cur.first, cur.second, bid});
    bcur[bid] = cur;
}
void setnode(int u) {
    for (auto &[c,shc, dit] : dist[u]) {
        bpq[shc].push({dit, u});
        sync(c, shc);
    }
}
void remnode(int u) {
    for (auto &[c,shx, dit] : dist[u]) {
        sync(c,shx);
    }
}
pair<int, int> getans(int u) {
    pair<int, int> rt = {0,u};
    if (!chck[u].empty())
        rt = max(rt, {get<0>(*chck[u].rbegin()), get<1>(*chck[u].rbegin())});
    for (auto &[c, exc, dit] : dist[u]) {
        for (auto it = chck[c].rbegin(); it != chck[c].rend(); it++) {
            if (cl[c] == 0)
            rt = max(rt, {dit, c});
            auto &[distn, node, excme] = *it;
            if (excme == exc) continue;
            rt = max(rt, {distn + dit, node});
            break;
        }
    }
    return rt;
}

    fio
    int t;
    cin >> t;
    while (t--) {
        int q;
        cin >> n >> q;
        for (int p, i = 1; i < n; i++) {
            cin >>  p; --p;
            g[p].push_back(i);
            g[i].push_back(p);
        }
        initc();
        for (int i = 0; i < n; i++) setnode(i);//, reverse(dist[i].begin(), dist[i].end());
       // vector<int> cl(n+1);
        for (int v, i = 0; i < q; i++) {
            cin >> v; v--;
            if (cl[v]) {//1, black
                cl[v] = 0;
                setnode(v);
                cout <<  getans(v).second + 1;
            }else {//0, white
                cout << getans(v).second +  1;
                cl[v] = 1;
                remnode(v);
            }
            cout << el;
        }
        for (int i = 0; i <= n; i++)
            cl[i] = 0,
        chck[i].clear(), vis2[i] = 0, g[i].clear(), sz[i] =  0 , vis[i] = 0, dist[i].clear();
        for (int i = 0; i <= id; i++) {
            bcur[i] =  {-1,-1};
            while (bpq[i].size()) bpq[i].pop();
        }
        id = 0;
    }
    return 0;
}
