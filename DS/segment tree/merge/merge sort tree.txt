#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef long double ld;
#define el '\n'
#define fio ios_base::sync_with_stdio(0); cin.tie(0); cout.tie(0);
const int N = 200'123;
int a[N];
struct Node {
    vector<int> srt;
};
Node create(int n) {
    return Node{{n}};
}
Node add(Node a, Node b) {
    vector<int> v; v.reserve(a.srt.size() + (int)b.srt.size());
    merge(a.srt.begin(), a.srt.end(), b.srt.begin(), b.srt.end(), back_inserter(v));
    return Node{v};
}
struct ST{
    Node t[N << 2];
    ST(){};//memset(t, 0, sizeof(t));}
    void build(int n, int b, int e){
        if(b == e) return void(t[n] = create(a[b]));//////////////////////
        int m = (b + e) >> 1, l = n << 1, r = l | 1;
        build(l, b, m);
        build(r, m + 1, e);
        t[n] = add(t[l], t[r]);
    }
    void upd(int n, int b, int e, int i){
        if(b > i || e < i) return;
        if(b == e) return void(t[n] = create(a[i]));
        int m = (b + e) >> 1, l = n << 1, r = l|1;
        upd(l, b, m, i);
        upd(r, m + 1, e, i);
        t[n] = add(t[l], t[r]);
    }
    int query(int n, int b, int e, int i, int j, int k){
        if(e < i || b > j) return 0;
        if(i <= b && e <= j) {
            auto &v = t[n].srt;
            int l{}, r = v.size()-1, md, ans = v.size();
            while (l <= r) {
                md = l+r>>1;
                if (v[md]<=k) {
                    l = md + 1;
                }
                else r = md - 1, ans = md;
            }
            return (int)v.size() - ans;
        }
        int m = (b + e) >> 1, l = n << 1, r = l|1;
        return query(l, b, m, i, j, k) + query(r, m + 1, e, i, j, k);
    }
}t;

void solve() {
    int n, q;
    cin >> n;
    for (int i = 1; i <= n; i++)
        cin >> a[i];
    t.build(1, 1, n);
    cin >> q;
    for (int i,j,k; q--;) {
        cin >>  i >>  j >> k;
        cout << t.query(1,1,n, i, j, k) << el;
    }

}

signed main() { fio int t = 1; /*cin >> t;*/ while (t--) solve(); return 0; }
