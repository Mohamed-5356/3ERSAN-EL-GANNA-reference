#include <bits/stdc++.h>
using namespace std;
typedef long long ll;
typedef long double ld;
#define el '\n'
#define fio ios_base::sync_with_stdio(0); cin.tie(0); cout.tie(0);
const int N = 2e5+9;
int tin[N], low[N], dfn, compid[N], compsz[N], id;
bool instack[N];
stack<int> st;
vector<int> g[N], comp[N], dag[N];
void tarjan(int u) {
    tin[u] = low[u] = ++dfn;
    st.push(u);
    instack[u] = 1;
    for (auto &v : g[u]) {
        if (tin[v]==0) {
            tarjan(v);
            low[u] = min(low[u], low[v]);
        }
        else if (instack[v]) {
            low[u] = min(low[u], tin[v]);
        }
    }
    if (tin[u] == low[u]) {
        while (true) {
            int x = st.top(); st.pop();
            instack[x] = false;
            compid[x] = id;
            comp[id].push_back(x);
            compsz[id]++;
            if (x == u) break;
        }
        id++;
    }
}
void builddag(int n){
    for (int u = 1; u <= n; u++) {
        for (auto &v : g[u])
            if (compid[u]!=compid[v])
                dag[compid[u]].push_back(compid[v]);
    }
    for (int c = 0; c < id; c++) {
        sort(dag[c].begin(), dag[c].end());
        dag[c].erase(unique(dag[c].begin(), dag[c].end()), dag[c].end());
    }

}
void solve() {
    int n , m; cin >> n >> m;
    for (int u, v, i = 0; i < m; i++) {
        cin >> u >> v;
        g[u].push_back(v);
    }
    for (int i = 1; i <= n; i++)
        if (!tin[i]) tarjan(i);
    builddag(n);
}





signed main() { fio int t = 1; /*cin >> t;*/ while (t--) solve(); return 0; }
