#include <bits/stdc++.h>
typedef long long ll;
#define el '\n'
#define fio ios_base::sync_with_stdio(0); cin.tie(0); cout.tie(0);
using namespace std;
vector<int> kmp(string s) {
    int n = s.size();
    vector<int> pi(n);
    for (int i = 1; i < n; i++) {
        int j = pi[i-1];
        while (j && s[i] != s[j])
            j = pi[j-1];
        if (s[i] == s[j])
            j++;
        pi[i] = j;
    }
    return pi;
}
vector<int> Zalgo(string &s) {
    int n = s.size();
    vector<int> z(n);
    int l = 0, r = 0;
    for (int i = 1; i < n; i++) {
        if (i < r) {
            z[i] = min(r - i, z[i - l]);
        }
        while (i + z[i] < n && s[z[i]] == s[i + z[i]]) {
            z[i]++;
        }
        if (i + z[i] > r) {
            l = i, r = i + z[i];
        }
    }
    return z;
}
void solve(){
    string s;
    cin >> s;
    auto v2 = kmp(s);
    auto v  = Zalgo(s);
    for (auto &i : v)
        cout << i << ' ';
    cout << el;
    for (auto &i : v2)
        cout << i << ' ';
    cout << el;
}

signed main(){ fio ll t = 1; /*cin >> t;*/ while(t--) solve(); return 0; }
